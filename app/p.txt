'use client';

import { useState } from 'react';

export default function FlatlayGeneratorPage() {
  const [selectedImage, setSelectedImage] = useState<string | null>(null);
  const [mimeType, setMimeType] = useState<string>('image/jpeg');
  const [loading, setLoading] = useState(false);
  const [flatlayPrompt, setFlatlayPrompt] = useState('');

  // Xử lý khi người dùng chọn ảnh
  const handleImageChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      setMimeType(file.type);
      const reader = new FileReader();
      reader.onloadend = () => {
        setSelectedImage(reader.result as string);
      };
      reader.readAsDataURL(file);
    }
  };

  // Gửi ảnh lên API để Gemini phân tích
  const handleAnalyze = async () => {
    if (!selectedImage) return;

    setLoading(true);
    setFlatlayPrompt('');

    try {
      const res = await fetch('/api/generate-flatlay', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ imageBase64: selectedImage, mimeType }),
      });

      const data = await res.json();
      if (res.ok) {
        setFlatlayPrompt(data.prompt);
      } else {
        setFlatlayPrompt(`Lỗi: ${data.error}`);
      }
    } catch (err) {
      setFlatlayPrompt('Đã xảy ra lỗi kết nối.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <main className="max-w-2xl mx-auto p-6 font-sans">
      <h1 className="text-3xl font-bold mb-2 text-center">
        AI Flatlay Prompt Generator
      </h1>
      <p className="text-gray-600 text-center mb-6">
        Tải ảnh sản phẩm của bạn lên để Gemini tạo prompt chụp flatlay chuyên
        nghiệp.
      </p>

      <div className="flex flex-col gap-4 border-2 border-dashed border-gray-300 p-6 rounded-xl text-center bg-gray-50">
        <input
          type="file"
          accept="image/*"
          onChange={handleImageChange}
          className="file:mr-4 file:py-2 file:px-4 file:rounded-full file:border-0 file:text-sm file:font-semibold file:bg-blue-50 file:text-blue-700 hover:file:bg-blue-100 cursor-pointer mx-auto"
        />

        {selectedImage && (
          <div className="mt-4 flex justify-center">
            <img
              src={selectedImage}
              alt="Product Preview"
              className="max-h-64 rounded-lg shadow-md object-contain"
            />
          </div>
        )}
      </div>

      <button
        onClick={handleAnalyze}
        disabled={!selectedImage || loading}
        className="mt-4 w-full bg-blue-600 text-white py-3 rounded-lg font-semibold hover:bg-blue-700 disabled:bg-gray-300 transition-colors"
      >
        {loading ? 'Gemini đang phân tích sản phẩm...' : 'Tạo Prompt Flatlay'}
      </button>

      {flatlayPrompt && (
        <div className="mt-6 p-4 bg-white border border-gray-200 rounded-xl shadow-sm">
          <h3 className="font-bold text-lg mb-2 text-gray-800">
            Prompt Flatlay được tạo:
          </h3>
          <p className="bg-gray-100 p-3 rounded-md text-sm text-gray-700 whitespace-pre-wrap font-mono select-all">
            {flatlayPrompt}
          </p>
          <button
            onClick={() => navigator.clipboard.writeText(flatlayPrompt)}
            className="mt-3 text-sm bg-gray-200 hover:bg-gray-300 text-gray-800 px-3 py-1.5 rounded font-medium transition"
          >
            📋 Sao chép Prompt
          </button>
        </div>
      )}
    </main>
  );
}

        // Must describe: background surface, curated complementary props, top-down angle, studio lighting, and high-end commercial aesthetics. 
        // Return ONLY the final prompt text.
